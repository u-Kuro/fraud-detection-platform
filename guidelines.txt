$ErrorActionPreference = "Stop"

$container = "ministack-test"
$bucket = "test-bucket"

try {
    Write-Host "Starting MiniStack..."

    docker run `
        --detach `
        --name $container `
        --publish 4566:4566 `
        ministackorg/ministack:1.5

    Write-Host "Waiting for MiniStack to become healthy..."

    $health = $null

    for ($i = 0; $i -lt 30; $i++) {
        $health = docker inspect `
            --format "{{.State.Health.Status}}" `
            $container 2>$null

        Write-Host "Health: $health"

        if ($health -eq "healthy") {
            Write-Host "MiniStack is healthy."
            break
        }

        if ($health -eq "unhealthy") {
            throw "MiniStack became unhealthy."
        }

        Start-Sleep -Seconds 1
    }

    if ($health -ne "healthy") {
        throw "MiniStack did not become healthy within 30 seconds."
    }

    # Create S3 bucket
    Write-Host "Creating bucket: $bucket"

    aws `
        --endpoint-url=http://localhost:4566 `
        s3 mb "s3://$bucket"

    # Create CSV
    @"
id,name,value
1,Alice,100
2,Bob,200
3,Charlie,300
"@ | Set-Content -NoNewline data.csv

    Write-Host ""
    Write-Host "Original CSV:"
    Write-Host "-------------"
    Get-Content data.csv
    Write-Host "-------------"

    # Upload CSV
    Write-Host ""
    Write-Host "Uploading CSV..."

    aws `
        --endpoint-url=http://localhost:4566 `
        s3 cp data.csv `
        "s3://$bucket/data.csv"

    # Download CSV
    Write-Host ""
    Write-Host "Downloading CSV..."

    aws `
        --endpoint-url=http://localhost:4566 `
        s3 cp "s3://$bucket/data.csv" `
        downloaded.csv

    Write-Host ""
    Write-Host "Downloaded CSV:"
    Write-Host "---------------"
    Get-Content downloaded.csv
    Write-Host "---------------"

    # Compare exact contents
    $original = Get-Content data.csv -Raw
    $downloaded = Get-Content downloaded.csv -Raw

    Write-Host ""
    Write-Host "Original raw value:"
    Write-Host "<$original>"

    Write-Host ""
    Write-Host "Downloaded raw value:"
    Write-Host "<$downloaded>"

    Write-Host ""
    Write-Host "Original byte count:   $([System.Text.Encoding]::UTF8.GetByteCount($original))"
    Write-Host "Downloaded byte count: $([System.Text.Encoding]::UTF8.GetByteCount($downloaded))"

    if ($original -ne $downloaded) {
        throw "Downloaded CSV does not match the original."
    }

    Write-Host ""
    Write-Host "SUCCESS: CSV values match exactly."
}
finally {
    Write-Host ""
    Write-Host "Cleaning up MiniStack..."

    docker rm --force $container 2>$null

    Write-Host "MiniStack container cleaned up."
}
